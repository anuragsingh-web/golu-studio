function GameCard({ c, rank, note }: { c: Card; rank: number; note?: string }) {
  const t = useGrokTheme();
  const bar = (label: string, v?: number) => v === undefined ? null : (
    <div key={label} style={{ display: "grid", gridTemplateColumns: "74px 1fr 30px", alignItems: "center", gap: 8 }}>
      <span style={{ ...t.font.caption, color: t.text.tertiary }}>{label}</span>
      <div style={{ height: 6, borderRadius: 3, background: t.border.subtle, overflow: "hidden" }}>
        <div style={{ width: `${v * 10}%`, height: "100%", borderRadius: 3, background: t.dataviz.categorical[0] }} />
      </div>
      <span style={{ ...t.font.caption, color: t.text.secondary, textAlign: "right" }}>{v.toFixed(1)}</span>
    </div>
  );
  const facts: [string, string][] = [];
  if (c.dl30 !== undefined) facts.push(["Downloads 30d", fmtN(c.dl30)]);
  if (c.dlGrowth !== undefined) facts.push(["vs prior 30d", fmtG(c.dlGrowth)]);
  if (c.rev30 !== undefined && c.rev30 > 0) facts.push(["Store revenue 30d", "$" + fmtN(c.rev30)]);
  if (c.dau !== undefined) facts.push(["DAU", fmtN(c.dau)]);
  if (c.ageDays !== undefined) facts.push(["Age", `${c.ageDays.toLocaleString("en-US")} days`]);
  if (c.family) facts.push(["Mechanic", c.family + (c.saturated === false ? " (open wave)" : c.saturated ? " (saturated)" : "")]);
  if (c.input) facts.push(["Input", c.input]);
  if (c.confidence) facts.push(["Confidence", c.confidence]);
  return (
    <div style={{ display: "flex", flexDirection: "column", gap: 12 }}>
      <div style={{ display: "flex", gap: 12, alignItems: "center" }}>
        <div style={{ width: 56, height: 56, flex: "0 0 56px", borderRadius: t.radius.lg, overflow: "hidden", background: t.bg.subtle }}>
          {c.icon && ICON_SRC[c.icon] ? <Image src={ICON_SRC[c.icon]} alt={`${c.name} icon`} width={56} ratio="1/1" fit="cover" /> : null}
        </div>
        <div style={{ flex: "1 1 auto", minWidth: 0 }}>
          <div style={{ ...t.font.caption, color: t.text.tertiary }}>#{rank} · {c.platform}</div>
          <div style={{ ...t.font.heading3, fontWeight: t.weight.semibold, color: t.text.primary }}>{c.name}</div>
          {c.developer ? <div style={{ ...t.font.caption, color: t.text.tertiary }}>{c.developer}</div> : null}
        </div>
        <div style={{ textAlign: "right", flex: "0 0 auto" }}>
          <div style={{ ...t.font.metric, color: t.text.primary, fontWeight: t.weight.semibold }}>{c.overall.toFixed(1)}</div>
          <div style={{ ...t.font.caption, color: t.text.tertiary }}>Pintu /10</div>
        </div>
      </div>
      {c.why && !c.unique ? <div style={{ ...t.font.body2, color: t.text.secondary }}>{c.why}</div> : null}
      <div style={{ display: "flex", flexDirection: "column", gap: 6 }}>
        {bar("Momentum", c.momentum)}
        {bar("Staying", c.staying)}
        {bar("Market size", c.size)}
        {bar("Hook", c.hook)}
      </div>
      <div style={{ display: "grid", gridTemplateColumns: "repeat(2, minmax(0, 1fr))", gap: "6px 12px" }}>
        {facts.map(([k, v]: [string, string]) => (
          <div key={k} style={{ minWidth: 0 }}>
            <div style={{ ...t.font.caption, color: t.text.tertiary }}>{k}</div>
            <div style={{ ...t.font.label, color: t.text.primary }}>{v}</div>
          </div>
        ))}
      </div>
      {c.metrics ? <div style={{ ...t.font.caption, color: t.text.tertiary }}>{c.metrics}</div> : null}
      {note ? <div style={{ ...t.font.body2, color: t.text.primary, borderTop: `1px solid ${t.border.subtle}`, paddingTop: 10 }}>{note}</div> : null}
      {c.perf ? <div style={{ ...t.font.caption, color: t.text.secondary }}>{`Performance: ${c.perf}`}</div> : null}
      {c.cap ? <div style={{ ...t.font.caption, color: t.text.tertiary }}>{c.cap}</div> : null}
      {c.url ? <a href={c.url} target="_blank" rel="noreferrer" style={{ ...t.font.label, color: t.text.accent, textDecoration: "none" }}>Open store page ↗</a> : null}
    </div>
  );
}

type TabK = "overview" | "new" | "up" | "genre" | "charts" | "trends" | "all" | "method";
type TabOpt = { value: TabK; label: string; count?: number };

function TabBar({ tabs, value, onChange }: { tabs: TabOpt[]; value: TabK; onChange: (v: TabK) => void }) {
  const t = useGrokTheme();
  const accent = t.dataviz.categorical[0];
  return (
    <div style={{ padding: "12px 0 16px", borderBottom: `1px solid ${t.border.subtle}` }}>
    <div style={{ overflowX: "auto", scrollbarWidth: "none", WebkitOverflowScrolling: "touch", maxWidth: "100%" }}>
    <div role="tablist" aria-label="Mood board sections" style={{
      display: "inline-flex", flexWrap: "nowrap", gap: 4, padding: 4, width: "max-content",
      background: t.bg.subtle, border: `1px solid ${t.border.default}`, borderRadius: t.radius.full,
    }}>
      {tabs.map((o: TabOpt) => {
        const active = o.value === value;
        return (
          <button
            key={o.value}
            role="tab"
            aria-selected={active}
            onClick={() => onChange(o.value)}
            style={{
              display: "inline-flex", alignItems: "center", gap: 6, padding: "8px 14px", borderRadius: t.radius.full,
              border: "none", background: active ? accent : "transparent", whiteSpace: "nowrap",
              color: active ? t.text.onColor : t.text.secondary, fontFamily: t.font.family, fontSize: 14, lineHeight: "20px",
              fontWeight: active ? 700 : t.weight.semibold, cursor: "pointer", transition: t.motion.hover,
            }}
          >
            {o.label}
            {o.count !== undefined ? (
              <span style={{
                display: "inline-block", minWidth: 20, padding: "0 6px", borderRadius: t.radius.full, textAlign: "center",
                fontSize: 12, lineHeight: "18px", fontWeight: t.weight.semibold,
                background: active ? t.text.onColor : t.fill.secondary, color: active ? accent : t.text.secondary,
              }}>{o.count}</span>
            ) : null}
          </button>
        );
      })}
    </div>
    </div>
    </div>
  );
}
